\section{Contexto Institucional}
\begin{frame}[t]{Diacon\'ia FRIF-IFD y el Programa BUMAND}
    \vspace{-0.08cm}
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.58\textwidth}
            \justifying
            \textbf{Diacon\'ia FRIF-IFD} es una entidad financiera de desarrollo con presencia nacional en Bolivia (Oficina Central en El Alto y 62 agencias), orientada a la inclusi\'on financiera y el progreso integral bajo principios cristianos.

            \vspace{0.10cm}
            \textbf{El Programa BUMAND} (Becas Universitarias con Misi\'on y Acci\'on Diaconal):
            \begin{itemize}
                \setlength{\itemsep}{2pt}
                \item Subvenci\'on de estudios universitarios a j\'ovenes en vulnerabilidad (becas de Bs. 650 y Bs. 250).
                \item Pr\'acticas preprofesionales en unidades de Diacon\'ia con meta de \textbf{36 horas mensuales}.
                \item Escuela de L\'ideres (viernes) y servicio voluntario en \textbf{17 iglesias} y congregaciones cristianas.
            \end{itemize}
        \end{column}
        \begin{column}{0.39\textwidth}
            \centering
            \safeimage{width=0.60\linewidth,keepaspectratio}{logo-bumand.png}
            \vspace{0.10cm}

                \textbf{25 becarios activos} (m\'as de 65 hist\'oricos) acompa\~nados por un cat\'alogo de \textbf{29 supervisores} institucionales en todo el pa\'is.
            }
        \end{column}
    \end{columns}
\end{frame}

    \vspace{-0.16cm}
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.50\textwidth}
            \justifying
            Digitaliza la planilla de transporte urbano con reglas flexibles y c\'alculo exacto:

            \vspace{0.05cm}
            \begin{itemize}
                \setlength{\itemsep}{0.5pt}
                \item \textbf{Tarifario sin Deduplicaci\'on por Monto:} Permite m\'ultiples tarifas con igual monto (Minib\'us, Microb\'us, Trufi, Telef\'erico).

                \begin{center}
                    \scriptsize $\text{Subvenci\'on} = \left(\sum \text{Tramos Diarios}\right) \times 0.80$
                \end{center}
                \item \textbf{Ventana de Env\'io:} Habilitaci\'on ordenada desde el \textbf{d\'ia 24} de cada mes.
